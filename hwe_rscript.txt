### libraries
library(ggplot2)
library(data.table)

### 1. HWE Function
# Takes a vector of allele frequencies (p) and returns a WIDE data.table
hwe_fun <- function(p) {
  q <- 1 - p
  
  f_AA <- p^2
  f_Aa <- 2 * p * q
  f_aa <- q^2
  
  return(data.table(p = p, AA = f_AA, Aa = f_Aa, aa = f_aa))
}

### 2. Generate Data
# Create a sequence of p values from 0 to 1
hwe_wide <- hwe_fun(p = seq(0, 1, by = 0.01))

### 3. Convert WIDE to LONG data using melt()
# Transforms AA, Aa, aa columns into variable and value columns
hwe_long <- melt(
  hwe_wide, 
  id.vars = "p", 
  measure.vars = c("AA", "Aa", "aa"),
  variable.name = "Genotype",
  value.name = "Frequency"
)

### 4. Plot using ggplot2
hwe_plot <- ggplot(data = hwe_long) +
  geom_line(aes(x = p, y = Frequency, color = Genotype, linetype = Genotype), linewidth = 1.2) +
  scale_color_manual(values = c("AA" = "blue", "Aa" = "red", "aa" = "darkgreen")) +
  labs(
    title = "Hardy-Weinberg Equilibrium Genotype Frequencies",
    x = "Allele Frequency (p)",
    y = "Genotype Frequency"
  ) +
  theme_classic()

# Display plot
print(hwe_plot)
